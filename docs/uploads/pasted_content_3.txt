import importlib

REPO_URL = "https://github.com/solelyart/ai-image-tool.git"
INSTALL_DIR = "/home/ubuntu/logo_processor_toolkit_repo"
TOOLKIT_PATH = os.path.join(INSTALL_DIR, "logo_processor_toolkit")

def run_command(cmd, description=""):
    """Run a shell command and return success status."""
    print(f"  → {description}..." if description else f"  → Running: {cmd}")
    try:
        result = subprocess.run(
            cmd, 
            shell=True, 
            capture_output=True, 
            text=True,
            timeout=300
        )
        if result.returncode != 0:
            print(f"    Warning: {result.stderr.strip()}")
            return False
        return True
    except subprocess.TimeoutExpired:
        print(f"    Warning: Command timed out")
        return False
    except Exception as e:
        print(f"    Warning: {e}")
        return False

def install_dependencies():
    """Install required Python packages."""
    print("\n📦 Installing dependencies...")
    
    packages = [
        "numpy",
        "Pillow", 
        "scikit-learn",
        "opencv-python",
        "pyyaml"
    ]
    
    for pkg in packages:
        run_command(f"pip3 install -q {pkg}", f"Installing {pkg}")
    
    print("  ✓ Dependencies installed")

def clone_or_update_repo():
    """Clone the repository or update if it exists."""
    print("\n📥 Setting up repository...")
    
    if os.path.exists(INSTALL_DIR):
        # Update existing repo
        print(f"  → Repository exists at {INSTALL_DIR}")
        os.chdir(INSTALL_DIR)
        run_command("git fetch origin main", "Fetching updates")
        run_command("git reset --hard origin/main", "Updating to latest")
        print("  ✓ Repository updated")
    else:
        # Clone new repo
        run_command(f"git clone {REPO_URL} {INSTALL_DIR}", "Cloning repository")
        print("  ✓ Repository cloned")

def setup_python_path():
    """Add toolkit to Python path."""
    print("\n🔧 Configuring Python path...")
    
    if INSTALL_DIR not in sys.path:
        sys.path.insert(0, INSTALL_DIR)
        print(f"  ✓ Added {INSTALL_DIR} to Python path")
    else:
        print(f"  ✓ Path already configured")

def verify_installation():
    """Verify the toolkit can be imported."""
    print("\n✅ Verifying installation...")
    
    try:
        # Clear any cached imports
        modules_to_remove = [m for m in sys.modules if m.startswith('logo_processor_toolkit')]
        for m in modules_to_remove:
            del sys.modules[m]
        
        # Try importing
        from logo_processor_toolkit import (
            __version__,
            LogoProcessor,
            KMeansColorDetector,
            ColorPalette,
            get_preset,
            list_presets
        )
        
        print(f"  ✓ Logo Processor Toolkit v{__version__} ready!")
        print(f"  ✓ Available presets: {list_presets()}")
        return True
        
    except ImportError as e:
        print(f"  ✗ Import failed: {e}")
        return False

def setup():
    """Main setup function."""
    print("="*60)
    print("🚀 LOGO PROCESSOR TOOLKIT - MANUS SETUP")

    install_dependencies()
    clone_or_update_repo()
